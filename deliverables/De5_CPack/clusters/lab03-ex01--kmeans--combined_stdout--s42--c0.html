{
	int N = 0;

	scanf("%d", &N);
	while (N <= 2);
	quadrado(N);
	return 0;
}

void quadrado(int N)
{	int valor, digito, contador;
	for (digito = 1; digito <= N; digito++)
	{	valor = digito;
		contador = 1;
		do{
			printf("%d ", valor);
			valor++;
			contador++;
		} while (contador <= N);
		putchar('\n');
	}
}

int main(){
	int n;
	scanf("%d", &n);
	
	while(n < 2){
		scanf("%d", &n);
	}
	quadrado(n);
	return 0;
}

#include <stdio.h>

void quadrado(int n)
{
    int i, j;
    for (i = 1; i <= n; i++){
        for (j = 0; j < n; j++){
            if (j) putchar('\t');
            printf("%d\t", i + j);
        }
        putchar('\n');
    }
}
int main()
{
    int n;
    scanf("%d", &n);
    if (n < 2) {
        printf("Número inválido!\n");
        return 1;
    }
    quadrado(n);
    return 0;
}
