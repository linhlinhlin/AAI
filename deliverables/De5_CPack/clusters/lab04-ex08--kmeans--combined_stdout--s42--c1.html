    n1 = transnum(s1);
    n2 = transnum(s2);

        printf("%s\n", s1);

long transnum(char s[]){              
    int i, comp = 0, alg;
    long num = 0;

    for(i=0; s[i] != '\0'; i++)
        comp++;

    for(i=0; s[i] != '\0'; i++){
        alg = s[i] - '0';
        num = num +  (alg * pot(comp));
        comp--;
    }
    return num;  
}

int pot(int e){
    int p = 1, b = 10;
    if(e == 0)
        return 1;
    else{
    while(e > 0){
        p = p*b;
        e--;
        }
    }
    return p;
}

void maior_numero(char s1[MAX], char s2[MAX]){
    int i;

    if(s1[0] == '-' && s2[0] != '-')
        printf("%s\n", s2);

    else if(s1[0] != '-' && s2[0] == '-')
        printf("%s\n", s1);

    else if(s1[0] == '-' && s2[0] == '-'){
        if(strlen(s1) < strlen(s2))
            printf("%s\n", s1);
        
        else if(strlen(s1) > strlen(s2))
            printf("%s\n", s2);

        else{
            for(i = 0; s1[i] != '\0'; i++){
                if(s1[i] > s2[i])
                    printf("%s\n", s2);
                if(s1[i] < s2[i])
                    printf("%s\n", s1);
            }
        }
    }
    
    else{
        if(strlen(s1) < strlen(s2))
            printf("%s\n", s2);
        
        else if(strlen(s1) > strlen(s2))
            printf("%s\n", s1);

        else{
            for(i = 0; s1[i] != '\0'; i++){
                if(s1[i] > s2[i])
                    printf("%s\n", s1);
                if(s1[i] < s2[i])
                    printf("%s\n", s2);
            }
        }
    }
}
